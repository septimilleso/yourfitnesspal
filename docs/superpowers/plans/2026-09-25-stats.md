`api/app/services/activity.py`:

```python
"""App-open activity days and the "days in a row" streak."""

from collections.abc import Iterable
from dataclasses import dataclass
from datetime import date, datetime, timedelta
from zoneinfo import ZoneInfo

import sqlalchemy as sa
from sqlalchemy.dialects.postgresql import insert
from sqlalchemy.ext.asyncio import AsyncSession

from app.config import settings
from app.models.activity import UserActivityDay
from app.models.user import User


@dataclass(frozen=True)
class Streak:
    current: int
    best: int


def local_today() -> date:
    return datetime.now(ZoneInfo(settings.app_timezone)).date()


def compute_streak(days: Iterable[date], today: date) -> Streak:
    """`current` is the run ending today, or yesterday when today is not
    recorded yet; `best` is the longest run ever. Days after `today` are
    ignored."""
    ordered = sorted({day for day in days if day <= today})
    best = 0
    run = 0
    previous: date | None = None
    for day in ordered:
        run = run + 1 if previous is not None and day - previous == timedelta(days=1) else 1
        best = max(best, run)
        previous = day
    current = run if previous is not None and today - previous <= timedelta(days=1) else 0
    return Streak(current=current, best=best)


async def record_activity(db: AsyncSession, user: User, day: date) -> None:
    await db.execute(
        insert(UserActivityDay)
        .values(user_id=user.id, day=day)
        .on_conflict_do_nothing(index_elements=["user_id", "day"])
    )
    await db.commit()


async def get_streak(db: AsyncSession, user: User, today: date) -> Streak:
    days = await db.scalars(
        sa.select(UserActivityDay.day).where(UserActivityDay.user_id == user.id)
    )
    return compute_streak(days.all(), today)
```

```python
MAX_RANGE_DAYS = 366


async def get_range(db: AsyncSession, user: User, start: date, end: date) -> WeekResponse:
    """Per-day totals for every day in `start..end` (inclusive), empty days
    included with zero totals."""
    if end < start:
        raise Invalid("end must not be before start")
    length = (end - start).days + 1
    if length > MAX_RANGE_DAYS:
        raise Invalid(f"range must not exceed {MAX_RANGE_DAYS} days")
    stmt = (
        sa.select(DiaryEntry)
        .where(DiaryEntry.user_id == user.id, DiaryEntry.date >= start, DiaryEntry.date <= end)
        .order_by(DiaryEntry.date)
    )
    entries = (await db.execute(stmt)).scalars().all()
    entries_by_date: dict[date, list[DiaryEntry]] = {
        start + timedelta(days=i): [] for i in range(length)
    }
    for entry in entries:
        entries_by_date[entry.date].append(entry)
    days = [
        WeekDayTotal(date=day, total=compute_day_totals(day_entries)["total"])
        for day, day_entries in sorted(entries_by_date.items())
    ]
    return WeekResponse(start=start, end=end, days=days)


async def get_week(db: AsyncSession, user: User, start: date) -> WeekResponse:
    return await get_range(db, user, start, start + timedelta(days=6))
```

```ts
import type { Goal, WeekDayTotal } from "../api/queries";
import { formatIsoDateShort, shiftIsoDate, startOfWeek } from "./date";
import { normalizeMacroPercentages } from "./goalMath";

export type Preset = 7 | 30 | 90;
export const PRESETS: Preset[] = [7, 30, 90];
export const MAX_RANGE_DAYS = 366;
/** Beyond this many days the charts show one bar per week. */
const DAILY_POINTS_LIMIT = 31;

export interface Period {
  start: string;
  end: string;
}

export const MACRO_SERIES: {
  key: "protein_g" | "carbs_g" | "fat_g";
  split: keyof MacroSplit;
  label: string;
  color: string;
}[] = [
  { key: "protein_g", split: "protein", label: "Proteine", color: "#3b82f6" },
  { key: "carbs_g", split: "carbs", label: "Carboidrati", color: "#f59e0b" },
  { key: "fat_g", split: "fat", label: "Grassi", color: "#a855f7" },
];

export function presetPeriod(days: Preset, today: string): Period {
  return { start: shiftIsoDate(today, -(days - 1)), end: today };
}

function isoToUtcMs(iso: string): number {
  const [year, month, dayOfMonth] = iso.split("-").map(Number);
  return Date.UTC(year, month - 1, dayOfMonth);
}

export function countDays(start: string, end: string): number {
  return Math.round((isoToUtcMs(end) - isoToUtcMs(start)) / 86_400_000) + 1;
}

/** Error message for an invalid custom range, or `null` when it is valid. */
export function validateCustomPeriod(from: string, to: string, today: string): string | null {
  if (!from || !to) return "Scegli entrambe le date.";
  if (from > to) return "La data iniziale deve precedere quella finale.";
  if (to > today) return "Le date non possono essere nel futuro.";
  if (countDays(from, to) > MAX_RANGE_DAYS) return "L'intervallo può coprire al massimo 366 giorni.";
  return null;
}

export interface ChartPoint {
  key: string;
  label: string;
  kcal: number;
  protein_g: number;
  carbs_g: number;
  fat_g: number;
  logged: boolean;
}

export interface Averages {
  kcal: number;
  protein_g: number;
  carbs_g: number;
  fat_g: number;
}

function isLogged(day: WeekDayTotal): boolean {
  return day.total.kcal > 0;
}

function average(days: WeekDayTotal[]): Averages | null {
  const logged = days.filter(isLogged);
  if (logged.length === 0) return null;
  const mean = (pick: (day: WeekDayTotal) => number) =>
    Math.round(logged.reduce((sum, day) => sum + pick(day), 0) / logged.length);
  return {
    kcal: mean((day) => day.total.kcal),
    protein_g: mean((day) => day.total.protein_g),
    carbs_g: mean((day) => day.total.carbs_g),
    fat_g: mean((day) => day.total.fat_g),
  };
}

function toPoint(key: string, days: WeekDayTotal[]): ChartPoint {
  const averages = average(days);
  return {
    key,
    label: formatIsoDateShort(key),
    kcal: averages?.kcal ?? 0,
    protein_g: averages?.protein_g ?? 0,
    carbs_g: averages?.carbs_g ?? 0,
    fat_g: averages?.fat_g ?? 0,
    logged: averages !== null,
  };
}

/** One point per day up to 31 days; beyond that one point per ISO week
 * (keyed by its Monday) with the average of that week's logged days. */
export function chartPoints(days: WeekDayTotal[]): ChartPoint[] {
  if (days.length <= DAILY_POINTS_LIMIT) {
    return days.map((day) => toPoint(day.date, [day]));
  }
  const weeks = new Map<string, WeekDayTotal[]>();
  for (const day of days) {
    const monday = startOfWeek(day.date);
    weeks.set(monday, [...(weeks.get(monday) ?? []), day]);
  }
  return [...weeks.entries()].map(([monday, weekDays]) => toPoint(monday, weekDays));
}

export interface PeriodSummary {
  totalDays: number;
  loggedDays: number;
  averages: Averages | null;
  daysOverGoal: number | null;
}

export function summarize(days: WeekDayTotal[], goal: Goal | null): PeriodSummary {
  const logged = days.filter(isLogged);
  return {
    totalDays: days.length,
    loggedDays: logged.length,
    averages: average(days),
    daysOverGoal: goal ? logged.filter((day) => day.total.kcal > goal.kcal).length : null,
  };
}

export interface MacroSplit {
  protein: number;
  carbs: number;
  fat: number;
}

/** Calorie share of each macro (4/4/9 kcal per gram) as integer percentages
 * summing to 100, or `null` when all macros are zero. */
export function macroSplit(grams: { protein_g: number; carbs_g: number; fat_g: number }): MacroSplit | null {
  const split = normalizeMacroPercentages(grams);
  if (split.protein === null || split.carbs === null || split.fat === null) return null;
  return { protein: split.protein, carbs: split.carbs, fat: split.fat };
}
```

```tsx
import { useQueryClient } from "@tanstack/react-query";
import { useEffect } from "react";

import { recordActivity } from "../api/queries";

/** Tells the API the app was opened today (streak counter): once on mount
 * and each time the app comes back to the foreground, which also covers a
 * PWA left open across midnight. Failures are ignored; the next foreground
 * retries. */
function ActivityTracker() {
  const queryClient = useQueryClient();

  useEffect(() => {
    function ping() {
      recordActivity()
        .then(() => queryClient.invalidateQueries({ queryKey: ["streak"] }))
        .catch(() => {});
    }
    function onVisibilityChange() {
      if (document.visibilityState === "visible") ping();
    }
    ping();
    document.addEventListener("visibilitychange", onVisibilityChange);
    return () => document.removeEventListener("visibilitychange", onVisibilityChange);
  }, [queryClient]);

  return null;
}

export default ActivityTracker;
```

```tsx
import type { ChartPoint } from "../../lib/stats";

const WIDTH = 320;
const HEIGHT = 160;
const TOP = 12;
const BOTTOM = 20;
const PLOT_HEIGHT = HEIGHT - TOP - BOTTOM;
const UNDER_GOAL = "#16a34a";
const OVER_GOAL = "#dc2626";
const NO_GOAL = "#94a3b8";

/** Every Nth label so at most ~7 fit under the bars. */
export function labelStep(count: number): number {
  return Math.max(1, Math.ceil(count / 7));
}

/** Daily (or weekly) kcal bars with a dashed goal line. */
function KcalChart({ points, goalKcal }: { points: ChartPoint[]; goalKcal: number | null }) {
  const hasData = points.some((point) => point.logged);
  const max = Math.max(goalKcal ?? 0, ...points.map((point) => point.kcal), 1) * 1.1;
  const slot = WIDTH / Math.max(points.length, 1);
  const barWidth = Math.max(slot * 0.7, 1);
  const y = (value: number) => TOP + PLOT_HEIGHT - (value / max) * PLOT_HEIGHT;
  const step = labelStep(points.length);

  return (
    <section className="card space-y-2 px-4 py-3">
      <div className="flex items-baseline justify-between">
        <h2 className="section-label">Calorie</h2>
        {goalKcal !== null && <span className="text-xs text-slate-500">Obiettivo {goalKcal} kcal</span>}
      </div>
      {!hasData ? (
        <p className="py-8 text-center text-sm text-slate-500">Nessun dato nel periodo</p>
      ) : (
        <svg viewBox={`0 0 ${WIDTH} ${HEIGHT}`} className="h-44 w-full" role="img" aria-label="Calorie per periodo">
          {points.map((point, index) =>
            point.logged ? (
              <rect
                key={point.key}
                data-testid="kcal-bar"
                x={index * slot + (slot - barWidth) / 2}
                y={y(point.kcal)}
                width={barWidth}
                height={TOP + PLOT_HEIGHT - y(point.kcal)}
                rx={Math.min(3, barWidth / 2)}
                fill={goalKcal === null ? NO_GOAL : point.kcal > goalKcal ? OVER_GOAL : UNDER_GOAL}
              >
                <title>{`${point.label}: ${point.kcal} kcal`}</title>
              </rect>
            ) : null,
          )}
          {goalKcal !== null && (
            <line
              data-testid="goal-line"
              x1={0}
              x2={WIDTH}
              y1={y(goalKcal)}
              y2={y(goalKcal)}
              stroke="#475569"
              strokeWidth={1}
              strokeDasharray="4 3"
            />
          )}
          {points.map((point, index) =>
            index % step === 0 ? (
              <text
                key={point.key}
                x={index * slot + slot / 2}
                y={HEIGHT - 4}
                textAnchor="middle"
                fontSize={9}
                fill="#64748b"
              >
                {point.label}
              </text>
            ) : null,
          )}
        </svg>
      )}
    </section>
  );
}

export default KcalChart;
```

```tsx
import { type ChartPoint, MACRO_SERIES } from "../../lib/stats";
import { labelStep } from "./KcalChart";

const WIDTH = 320;
const HEIGHT = 160;
const TOP = 12;
const BOTTOM = 20;
const PLOT_HEIGHT = HEIGHT - TOP - BOTTOM;

/** Stacked protein/carbs/fat grams per day (or week). */
function MacroChart({ points }: { points: ChartPoint[] }) {
  const hasData = points.some((point) => point.logged);
  const max =
    Math.max(1, ...points.map((point) => point.protein_g + point.carbs_g + point.fat_g)) * 1.1;
  const slot = WIDTH / Math.max(points.length, 1);
  const barWidth = Math.max(slot * 0.7, 1);
  const scale = (grams: number) => (grams / max) * PLOT_HEIGHT;
  const step = labelStep(points.length);

  return (
    <section className="card space-y-2 px-4 py-3">
      <h2 className="section-label">Macro (g)</h2>
      {!hasData ? (
        <p className="py-8 text-center text-sm text-slate-500">Nessun dato nel periodo</p>
      ) : (
        <>
          <svg viewBox={`0 0 ${WIDTH} ${HEIGHT}`} className="h-44 w-full" role="img" aria-label="Macro per periodo">
            {points.map((point, index) => {
              if (!point.logged) return null;
              let base = TOP + PLOT_HEIGHT;
              return (
                <g key={point.key}>
                  {MACRO_SERIES.map((series) => {
                    const height = scale(point[series.key]);
                    base -= height;
                    return (
                      <rect
                        key={series.key}
                        data-testid="macro-segment"
                        x={index * slot + (slot - barWidth) / 2}
                        y={base}
                        width={barWidth}
                        height={height}
                        fill={series.color}
                      >
                        <title>{`${point.label} · ${series.label}: ${point[series.key]} g`}</title>
                      </rect>
                    );
                  })}
                </g>
              );
            })}
            {points.map((point, index) =>
              index % step === 0 ? (
                <text
                  key={point.key}
                  x={index * slot + slot / 2}
                  y={HEIGHT - 4}
                  textAnchor="middle"
                  fontSize={9}
```

```tsx
import { PRESETS, type Preset } from "../../lib/stats";

export type PeriodChoice = { kind: "preset"; days: Preset } | { kind: "custom"; from: string; to: string };

interface Props {
  choice: PeriodChoice;
  today: string;
  error: string | null;
  onPreset: (days: Preset) => void;
  onCustom: () => void;
  onDates: (from: string, to: string) => void;
}

const BUTTON = "flex-1 rounded-md px-2 py-1.5 text-[13px] font-semibold transition";

/** 7 / 30 / 90 days or a custom from–to range. */
function PeriodPicker({ choice, today, error, onPreset, onCustom, onDates }: Props) {
  const active = (on: boolean) => (on ? "bg-white text-slate-900 shadow-sm" : "text-slate-500");
  return (
    <section className="card space-y-3 px-4 py-3">
      <div role="group" aria-label="Periodo" className="flex rounded-lg bg-slate-200/70 p-0.5">
        {PRESETS.map((days) => {
          const on = choice.kind === "preset" && choice.days === days;
          return (
            <button
              key={days}
              type="button"
              aria-pressed={on}
              onClick={() => onPreset(days)}
              className={`${BUTTON} ${active(on)}`}
            >
              {days} giorni
            </button>
          );
        })}
        <button
          type="button"
          aria-pressed={choice.kind === "custom"}
          onClick={onCustom}
          className={`${BUTTON} ${active(choice.kind === "custom")}`}
        >
          Personalizzato
        </button>
      </div>
      {choice.kind === "custom" && (
        <div className="grid grid-cols-2 gap-3">
          <label className="space-y-1 text-xs font-medium text-slate-600">
            <span>Dal</span>
            <input
              type="date"
              value={choice.from}
              max={today}
              onChange={(e) => onDates(e.target.value, choice.to)}
              className="min-h-[44px] w-full rounded-xl border border-slate-300 px-3 py-2 text-base"
            />
          </label>
          <label className="space-y-1 text-xs font-medium text-slate-600">
            <span>Al</span>
            <input
              type="date"
              value={choice.to}
              max={today}
              onChange={(e) => onDates(choice.from, e.target.value)}
              className="min-h-[44px] w-full rounded-xl border border-slate-300 px-3 py-2 text-base"
            />
          </label>
        </div>
      )}
      {error && (
        <p role="alert" className="text-sm text-red-600">
          {error}
        </p>
      )}
    </section>
  );
}

export default PeriodPicker;
```

```tsx
import { useSearchParams } from "react-router-dom";

import { useDiaryRange, useGoal } from "../api/queries";
import GoalForm from "../components/GoalForm";
import ModeSwitch from "../components/ModeSwitch";
import StreakCard from "../components/StreakCard";
import KcalChart from "../components/stats/KcalChart";
import MacroChart from "../components/stats/MacroChart";
import MacroSplitBars from "../components/stats/MacroSplitBars";
import PeriodPicker, { type PeriodChoice } from "../components/stats/PeriodPicker";
import SummaryTable from "../components/stats/SummaryTable";
import { todayIso } from "../lib/date";
import {
  chartPoints,
  macroSplit,
  type Period,
  type Preset,
  PRESETS,
  presetPeriod,
  summarize,
  validateCustomPeriod,
} from "../lib/stats";

function readChoice(params: URLSearchParams): PeriodChoice {
  const from = params.get("from");
  const to = params.get("to");
  if (from !== null || to !== null) return { kind: "custom", from: from ?? "", to: to ?? "" };
  const days = Number(params.get("range"));
  return { kind: "preset", days: (PRESETS as number[]).includes(days) ? (days as Preset) : 7 };
}

/** Diet stats: streak, goals, period picker, charts and averages, over the
 * chef-cat background (set by AppShell for /stats). The bottom spacer lets
 * the user scroll past the cards so the cat is fully visible. */
function DietStatsPage() {
  const [params, setParams] = useSearchParams();
  const today = todayIso();
  const choice = readChoice(params);
  const error = choice.kind === "custom" ? validateCustomPeriod(choice.from, choice.to, today) : null;
  const period: Period =
    choice.kind === "preset"
      ? presetPeriod(choice.days, today)
      : { start: choice.from, end: choice.to };

  const range = useDiaryRange(period.start, period.end, error === null);
  const { data: goal } = useGoal();
  const activeGoal = goal ?? null;

  const days = error === null ? (range.data?.days ?? []) : [];
  const points = chartPoints(days);
  const summary = summarize(days, activeGoal);

  function setPreset(daysCount: Preset) {
    setParams({ range: String(daysCount) }, { replace: true });
  }
  function openCustom() {
    setParams({ from: period.start, to: period.end }, { replace: true });
  }
  function setDates(from: string, to: string) {
    setParams({ from, to }, { replace: true });
  }

  return (
    <div className="stats-page">
      <header className="app-header">
        <ModeSwitch />
        <div className="mx-auto max-w-2xl px-4 py-3">
          <h1 className="text-[17px] font-semibold text-slate-900">Statistiche</h1>
        </div>
      </header>

      <div className="mx-auto max-w-2xl space-y-4 p-4">
        <StreakCard />
        <GoalForm />
        <PeriodPicker
          choice={choice}
          today={today}
          error={error}
          onPreset={setPreset}
          onCustom={openCustom}
          onDates={setDates}
        />
        {range.isError ? (
          <p role="alert" className="card px-4 py-3 text-sm text-red-600">
            Impossibile caricare le statistiche.
          </p>
        ) : range.isLoading && error === null ? (
          <p className="py-8 text-center text-sm text-slate-500">Caricamento...</p>
        ) : (
          <>
            <KcalChart points={points} goalKcal={activeGoal?.kcal ?? null} />
            <MacroChart points={points} />
            <MacroSplitBars
              consumed={summary.averages ? macroSplit(summary.averages) : null}
              target={activeGoal ? macroSplit(activeGoal) : null}
            />
            <SummaryTable summary={summary} />
          </>
        )}
      </div>
      <div aria-hidden="true" data-testid="cat-spacer" className="h-[55vh]" />
    </div>
  );
}

export default DietStatsPage;
```

Note: with `enabled: false`, TanStack Query v5 reports `isLoading: false`, so an invalid custom range shows the empty charts under the error. The `range.isLoading && error === null` guard keeps that explicit.
